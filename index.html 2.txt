    <script>
        async function apiCall(data) {
            try {
                const response = await fetch('/api/calculate', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(data)
                });
                return await response.json();
            } catch (error) {
                return { error: 'Erreur de connexion' };
            }
        }

        function showResult(elementId, result) {
            const el = document.getElementById(elementId);
            if (result.error) {
                el.className = 'result error';
                el.textContent = '❌ ' + result.error;
            } else {
                el.className = 'result success';
                el.textContent = '✓ ' + result.result;
            }
        }

        async function calculateBasic() {
            const a = parseFloat(document.getElementById('basic-a').value);
            const b = parseFloat(document.getElementById('basic-b').value);
            const op = document.getElementById('basic-op').value;
            
            const result = await apiCall({ operation: op, a, b });
            showResult('basic-result', result);
        }

        function toggleAdvancedInputs() {
            const op = document.getElementById('adv-op').value;
            const bGroup = document.getElementById('adv-b-group');
            const aLabel = document.getElementById('adv-a-label');
            const bLabel = document.getElementById('adv-b-label');
            
            if (op === 'sqrt') {
                bGroup.style.display = 'none';
                aLabel.textContent = 'Nombre';
            } else if (op === 'power') {
                bGroup.style.display = 'block';
                aLabel.textContent = 'Base';
                bLabel.textContent = 'Exposant';
            } else {
                bGroup.style.display = 'block';
                aLabel.textContent = 'A';
                bLabel.textContent = 'B';
            }
        }

        async function calculateAdvanced() {
            const a = parseFloat(document.getElementById('adv-a').value);
            const b = parseFloat(document.getElementById('adv-b').value);
            const op = document.getElementById('adv-op').value;
            
            let data;
            if (op === 'power') {
                data = { operation: op, base: a, exp: b };
            } else if (op === 'sqrt') {
                data = { operation: op, n: a };
            } else {
                data = { operation: op, a: Math.floor(a), b: Math.floor(b) };
            }
            
            const result = await apiCall(data);
            showResult('adv-result', result);
        }

        async function calculateNumber() {
            const n = parseInt(document.getElementById('num-n').value);
            const op = document.getElementById('num-op').value;
            
            const result = await apiCall({ operation: op, n });
            showResult('num-result', result);
        }

        // Initialize
        toggleAdvancedInputs();
    </script>
