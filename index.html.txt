        function showResult() {
            const name = document.getElementById('name').value || 'Anonyme';
            const language = document.getElementById('language').value;
            const languageNames = {
                'html': 'HTML',
                'python': 'Python',
                'cpp': 'C++'
            };
            
            const resultDiv = document.getElementById('result');
            resultDiv.style.display = 'block';
            resultDiv.innerHTML = `Bonjour <strong>${name}</strong>! Vous avez choisi <strong>${languageNames[language]}</strong> comme langage préféré.`;
